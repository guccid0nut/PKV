# Strip absolute PDB directory paths, enforce reproducible PE timestamps, and enable Control Flow Guard (/guard:cf) on MSVC Release builds
if(MSVC)
    add_compile_options("$<$<AND:$<COMPILE_LANGUAGE:C,CXX>,$<CONFIG:Release,RelWithDebInfo,MinSizeRel>>:/guard:cf>")
    add_link_options("/PDBALTPATH:%_PDB%" "/BREPRO" "$<$<CONFIG:Release,RelWithDebInfo,MinSizeRel>:/guard:cf>")
endif()

option(KONGOR_DEPLOY_TO_GAMES "Auto-deploy compiled native DLLs/.so/.dylib into games/* and artifacts/native-binaries/*" OFF)
